# Разделение трека на четыре слоя (drums/bass/other/vocals) — stems.py.
#
# ВАЖНО про версию Python. demucs и torch тянут скомпилированные колёса, и
# на самых свежих версиях Python их просто нет: на 3.14 установка падает
# на сборке зависимостей из исходников. Рабочий диапазон — 3.10-3.12, и он
# же нужен companion'у (у python-rtmidi колёс новее 3.12 тоже нет).
# Разбираться в этом руками не нужно: запустите setup_stems.ps1 из папки
# DARAVE — он найдёт подходящий Python, поставит туда всё, включая torch
# под вашу видеокарту, и скажет, что получилось.
#
# Здесь СОЗНАТЕЛЬНО только demucs. audio-separator (роформеры на вокал)
# ставится отдельной командой, потому что одна его зависимость
# (diffq-fixed) собирается из исходников и на новых Python падает — а pip
# при этом отменяет ВЕСЬ файл требований, и вместе с ней не ставится
# ничего, включая demucs. Ровно это и произошло.
#
#   pip install -r requirements-stems.txt
#   pip install audio-separator            # необязательно, для роформера
#
# demucs даёт четыре слоя моделью htdemucs_ft (дообученная htdemucs:
# заметно лучше базовой, считает вчетверо дольше).
demucs>=4.0.1
