# DARAVE — darave.ru. Apache (виртуальный хостинг).
#
# Если у вашего хостинга nginx без .htaccess — этот файл ничего не сломает,
# просто будет проигнорирован; тогда см. nginx.conf.example рядом.

# ------------------------------------------------------------ только HTTPS
# Порядок правил важен: сперва уводим на https, потом на www-без-www.
# Два отдельных редиректа вместо одного — чтобы не было цепочки из трёх
# переходов, которую поисковики считают за ошибку.
<IfModule mod_rewrite.c>
  RewriteEngine On

  # HTTPS. HTTP_X_FORWARDED_PROTO — для хостингов, где сертификат
  # заканчивается на балансировщике и до Apache доходит уже http.
  RewriteCond %{HTTPS} !=on
  RewriteCond %{HTTP:X-Forwarded-Proto} !https
  RewriteRule ^(.*)$ https://%{HTTP_HOST}/$1 [R=301,L]

  # Без www. Выберите одно написание и держитесь его: darave.ru и
  # www.darave.ru для поисковика — два разных сайта.
  RewriteCond %{HTTP_HOST} ^www\.(.+)$ [NC]
  RewriteRule ^(.*)$ https://%1/$1 [R=301,L]
</IfModule>

# -------------------------------------------------------------- установщик
# .exe должен отдаваться как файл для скачивания, а не как «неизвестный
# тип», который часть браузеров пытается открыть. И обязательно с
# докачкой: 173 МБ на мобильном интернете рвутся регулярно, и без
# Accept-Ranges человек начинает заново с нуля.
<IfModule mod_mime.c>
  AddType application/octet-stream .exe
</IfModule>
<IfModule mod_headers.c>
  <FilesMatch "\.exe$">
    Header set Accept-Ranges bytes
    Header set Content-Disposition "attachment"
    # Установщик версионный: имя файла меняется с версией, поэтому его
    # можно кэшировать надолго — перевыпуск придёт под новым именем.
    Header set Cache-Control "public, max-age=2592000, immutable"
  </FilesMatch>
</IfModule>

# ------------------------------------------------------------------ кэш
<IfModule mod_expires.c>
  ExpiresActive On
  # Страницу НЕ кэшируем надолго: правки текста и цен должны доезжать
  # сразу, а весит она 20 КБ.
  ExpiresByType text/html "access plus 10 minutes"
  ExpiresByType image/svg+xml "access plus 1 month"
  ExpiresByType image/png "access plus 1 month"
  ExpiresByType text/css "access plus 1 week"
  ExpiresByType application/javascript "access plus 1 week"
</IfModule>

<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript image/svg+xml
  # .exe уже сжат — повторное сжатие только греет процессор.
  SetEnvIfNoCase Request_URI \.exe$ no-gzip dont-vary
</IfModule>

# ---------------------------------------------------------------- защита
<IfModule mod_headers.c>
  Header always set X-Content-Type-Options "nosniff"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  # HSTS включайте ТОЛЬКО когда https уже работает и вы в нём уверены:
  # браузер запомнит запрет на http для этого домена на год, и откатить
  # это с вашей стороны нельзя — только ждать, пока истечёт.
  # Header always set Strict-Transport-Security "max-age=31536000"
</IfModule>

# Листинг каталогов выключаем: иначе /download/ показывает содержимое папки.
Options -Indexes

ErrorDocument 404 /index.html
